' =====================================
'          MODULE : Users
' =====================================
frame "UserModule" as F_User {
  class UserModule <<Module>> {
    - controllers: [UserController]
    - providers: [UserService, UserRepository]
  }

  class UserController <<Controller>> {
    + createUser(dto: CreateUserDto)
    + updateUser(id: number, dto: UpdateUserDto)
    + deleteUser(id: number)
    + findAll()
    + findOne(id: number)
  }

  class UserService <<Service>> {
    - userRepository: UserRepository
    + createUser(dto: CreateUserDto)
    + updateUser(id: number, dto: UpdateUserDto)
    + deleteUser(id: number)
    + findAll()
    + findOne(id: number)
  }

  class UserRepository <<Repository>> {
    + save(user: User)
    + findAll(): User[]
    + findById(id: number): User
    + remove(id: number)
  }

  class User <<Entity>> {
    + id: number
    + email: string
    + name: string
    + username: string
    + password: string
    + userCreationDate: Date
  }

  class CreateUserDto <<DTO>> {
    + email: string
    + name: string
    + username: string
    + password: string
  }

  class UpdateUserDto <<DTO>> {
    + email?: string
    + name?: string
    + username?: string
    + password?: string
  }

  UserModule --> UserController
  UserModule --> UserService
  UserModule --> UserRepository
  UserController --> UserService
  UserService --> UserRepository
  UserRepository --> User
  UserController ..> CreateUserDto : use
  UserController ..> UpdateUserDto : use
}

'        RELATIONS ENTRE ENTITES

' Basé sur le MCD (users owns land, land contains cultivation_spaces, etc.)

User "1" -- "0..*" Land : owns

CultivationSpace "1" -- "0..*" Crop : is cultivated
Crop "1" -- "0..*" SoilCover : is covered
