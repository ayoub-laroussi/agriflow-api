' =====================================
'         MODULE : Terrains
' =====================================
frame "TerrainModule" as F_Terrain {
  class TerrainModule <<Module>> {
    - controllers: [TerrainController]
    - providers: [TerrainService, TerrainRepository]
  }

  class TerrainController <<Controller>> {
   createTerrain(dto: CreateTerrainDto)
   updateTerrain(id: number, dto: UpdateTerrainDto)
   deleteTerrain(id: number)
   findAll()
   findOne(id: number)
  }

  class TerrainService <<Service>> {
    - terrainRepository: TerrainRepository
   createTerrain(dto: CreateTerrainDto)
   updateTerrain(id: number, dto: UpdateTerrainDto)
   deleteTerrain(id: number)
   findAll()
   findOne(id: number)
  }

  class TerrainRepository <<Repository>> {
   save(terrain: Terrain)
   findAll(): Terrain[]
   findById(id: number): Terrain
   remove(id: number)
  }

  class Terrain <<Entity>> {
   id: number
   nom: string
   surface: number
   uniteSurface: string
  }

  class CreateTerrainDto <<DTO>> {
   nom: string
   surface: number
   uniteSurface: string
  }

  class UpdateTerrainDto <<DTO>> {
   nom?: string
   surface?: number
   uniteSurface?: string
  }

  TerrainModule --> TerrainController
  TerrainModule --> TerrainService
  TerrainModule --> TerrainRepository
  TerrainController --> TerrainService
  TerrainService --> TerrainRepository
  TerrainRepository --> Terrain
  TerrainController ..> CreateTerrainDto : use
  TerrainController ..> UpdateTerrainDto : use
}

  class ActionAgricoleController <<Controller>> {
   createAction(dto: CreateActionDto)
   updateAction(id: number, dto: UpdateActionDto)
   deleteAction(id: number)

  }

  class ActionAgricoleService <<Service>> {
    - actionAgricoleRepository: ActionAgricoleRepository
   createAction(dto: CreateActionDto)
   updateAction(id: number, dto: UpdateActionDto)
   deleteAction(id: number)
   findAll()
   findOne(id: number)
   setPlantationDetails(actionId: number, spacingRow: number, spacingLine: number, nbPlants: number)
  }

  class ActionAgricoleRepository <<Repository>> {
   save(action: ActionAgricole)
   findAll(): ActionAgricole[]
   findById(id: number): ActionAgricole
   remove(id: number)
  }

  class ActionAgricole <<Entity>> {
   id: number
   type: string  ' (préparation, plantation, entretien, protection, récolte)
   dateAction: Date
   commentaire: string
  }

  class CreateActionDto <<DTO>> {
   type: string
   dateAction: Date
   commentaire?: string
   spacingRow?: number
   spacingLine?: number
   nbPlants?: number
  }

  class UpdateActionDto <<DTO>> {
   type?: string
   dateAction?: Date
   commentaire?: string
   spacingRow?: number
   spacingLine?: number
   nbPlants?: number
  }

  ActionAgricoleModule --> ActionAgricoleController
  ActionAgricoleModule --> ActionAgricoleService
  ActionAgricoleModule --> ActionAgricoleRepository
  ActionAgricoleController --> ActionAgricoleService
  ActionAgricoleService --> ActionAgricoleRepository
  ActionAgricoleRepository --> ActionAgricole
  ActionAgricoleController ..> CreateActionDto : use
  ActionAgricoleController ..> UpdateActionDto : use
}
